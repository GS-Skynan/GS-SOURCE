     void READ_NFC_SET_START(void)  // 
{
    I2C1_WriteNBytes(0x57,I2C_sendAdd_2,2);            //write system configuration //0x0000
    I2C1_ReadNBytes(0x57,I2C_R_system_config_Data,sizeof(I2C_R_system_config_Data));//read system configuration //0x0000
    delay(5);//0
    GPO_config_value=I2C_R_system_config_Data[0];
    delay(5);//1
    if(GPO_config_value==0XC0)
    {
      NOP();// RC2=1;
    }
    else          //----------Enable ST25DV04K GPO password------DIVECE-0X57---//
    {
        NOP();
        I2C1_WriteNBytes(0x57,I2C_password_add,19);
        delay(10);//5
        I2C1_WriteNBytes(0x57,I2C_sendAdd_2,3); 
        delay(2);//1
    }    
    I2C1_WriteNBytes(0x53,I2C_sendAdd,2);            //write NFC USE MEMORY address
    I2C1_ReadNBytes(0x53,I2C_receiveData,4);//read NFC USE MEMORY data //sizeof(I2C_receiveData
    read_eeprom_H=I2C_receiveData[2];
    read_eeprom=read_eeprom_H;
    read_eeprom=read_eeprom<<8;
    read_eeprom_L=I2C_receiveData[3];
    read_eeprom=read_eeprom+read_eeprom_L;
    delay(5);//10
}

{    
    if(GPO_INT_FLAG == 1)    //GPO INT(extern write eeprom  )  // pic16f18854-5 使用的是I/O口电平变化中断触发
    { 
      INTCONbits.GIE = 0;
      I2C1_WriteNBytes(0x53,I2C_sendAdd,2);            //write NFC USE MEMORY address
      I2C1_ReadNBytes(0x53,I2C_receiveData,4);//read NFC USE MEMORY data  sizeof I2C_receiveData
      delay(5);//
      read_eeprom_H=I2C_receiveData[2];
      read_eeprom=read_eeprom_H;
      read_eeprom=read_eeprom<<8;
      read_eeprom_L=I2C_receiveData[3];
      read_eeprom=read_eeprom+read_eeprom_L;
      GPO_INT_FLAG = 0 ; 
      INTCONbits.GIE = 1;
    }

void NFC_current_PWM(void)
{
  if(read_eeprom<=0X1850){read_eeprom=0X1850;} //limit NFC min_current_value
  if(read_eeprom>=0X2400){read_eeprom=0X2400;} //limit NFC max_current_value
  switch(read_eeprom)
       {
//*****************
      /*
       * 277-400-660W
       * v_out = 48-80v
       * current = 1.3-2.1A
       * 100mA/
       * VOUT_I  NFC current 
       * Ref_V_PWM  out_pwm
       * NFC_75_V  V_OUT 
       * 1850  
       * 2000
       * 2100
       * 2200
       * 2300
       * 2400
       */
       //******  
       case 0X1850: VOUT_I=0X1850;  break;//76.92V set=80v
        //******
       case 0X2000: VOUT_I=0X2000;  break;//71.43V set=75v
        //******
       case 0X2100: VOUT_I=0X2100;  break;//66.67V set=70v
       //******
       case 0X2200: VOUT_I=0X2200;  break;//62.5V set=65v
       //******
       case 0X2300: VOUT_I=0X2300;  break;//58.82V set=62v
       //******
       case 0X2400: VOUT_I=0X2400;  break;//55.56V set=58v
       // ******
       default: ;break;
        } 
}
